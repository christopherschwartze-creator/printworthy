# rtree: trimesh's nearest.on_surface() (the deviation-certificate two-sided
# Hausdorff measurement, used throughout the fixer) needs rtree to build its
# triangles_tree spatial index -- trimesh treats it as OPTIONAL and does not
# pull it in as a transitive dependency, so it must be listed explicitly.
# Its absence was a real, previously-undetected bug (2026-07-26): it never
# surfaced on the developer's own machine because rtree happened to already
# be installed globally there from unrelated past work, silently masking the
# gap -- it only showed up once tested in a genuinely clean environment,
# where every _fix_accurate case failed with worst_rms=inf (deviation
# certificate silently catching the ModuleNotFoundError and returning its
# worst-case default). MIT (rtree package + libspatialindex >=1.8.0, which
# all current wheels bundle).
rtree==1.4.1

# fast_simplification: trimesh.simplify_quadric_decimation is a thin wrapper
# that does `from fast_simplification import simplify` INSIDE the function
# body -- absent, it raises, and core._mesh_util.decimate()'s except-Exception
# fallback silently returns the mesh UNDECIMATED. This is the entire
# compute-safety cap (ingress guard's DECIMATE_FACES, the hosted Space's
# 20,000-face analysis proxy) -- not a nicety. SAME BUG CLASS as rtree above,
# a second time: never surfaced locally because it happened to already be
# installed globally (unrelated past work), silently masking the gap.
# Verified in production 2026-07-26: a real 267,873-face upload analyzed at
# FULL resolution instead of the intended 20,000-face cap, costing ~360s of a
# ~396s job (reproduced exactly by simulating its absence: decimate(mesh,
# 20000) returned all 267,873 faces unchanged). MIT.
fast-simplification==0.1.13

# Embree ray backend for trimesh: the thin-wall / premortem analysis casts
# one inward ray per face; pure-python casting is O(faces^2) and was the
# per-job time hotspot (measured locally on the worst corpus part:
# 278.6 s -> 136.1 s total with embree). cp310/cp311 manylinux_2_28 wheels
# verified available on PyPI for this exact version.
embreex==4.4.0
